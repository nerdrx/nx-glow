cmake_minimum_required(VERSION 3.25)
project(nx-glow VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KWin 6.7 REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6KCMUtils REQUIRED)
kcoreaddons_add_plugin(nxglow SOURCES src/glow.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
set_target_properties(nxglow PROPERTIES PREFIX "")
target_link_libraries(nxglow PRIVATE KWin::kwin)
target_compile_options(nxglow PRIVATE -Wall -Wextra)
kcoreaddons_add_plugin(nxglow_config SOURCES src/config.cpp INSTALL_NAMESPACE "kwin/effects/configs")
set_target_properties(nxglow_config PROPERTIES PREFIX "")
target_link_libraries(nxglow_config PRIVATE KF6::KCMUtils KF6::ConfigCore Qt6::Widgets Qt6::DBus)
option(NX_GLOW_BUILD_TESTS "Build native settings smoke test" OFF)
if(NX_GLOW_BUILD_TESTS)
    add_executable(config_smoke tests/config_smoke.cpp)
    target_link_libraries(config_smoke PRIVATE KF6::KCMUtils KF6::ConfigCore Qt6::Widgets)
endif()
